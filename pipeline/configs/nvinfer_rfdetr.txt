# WasteWatch Primary GIE — RF-DETR-Nano 2-class (person + handbag)
#
# Fine-tuned from RFDETR-Nano pretrained on COCO, retrained on 3.5k
# images (3000 COCO person/handbag-only + 528 LSG CCTV frames auto-
# labeled). Output layout: labels[N,300,3] (bg + 2 classes) + dets[N,300,4].
#
# Class index map (labels.txt lines):
#   0 person
#   1 handbag
# Training used class ids 0=person, 1=handbag. RFDETR's logits tensor
# has num_classes+1 channels but argmax-by-DeepStreamOutput returns
# the training index directly — so labels.txt must NOT have a
# background row or persons get shifted to "background" and suppressed.
#
# Input: 384×384 static (Nano default resolution). Keep infer-dims in
# lock-step with the ONNX export or DeepStream will silently resize and
# confuse the position-embedding layer.
#
# Copyright (c) 2024-2026 Satyaki Solutions Pvt Ltd. All rights reserved.

[property]
gpu-id=0
net-scale-factor=0.00392156862745

infer-dims=3;384;384

model-engine-file=/app/models/engine_cache/rfdetr_fp32.engine
onnx-file=/app/models/rfdetr/rfdetr.onnx
labelfile-path=/app/configs/labels.txt

batch-size=1
process-mode=1
model-color-format=0
# FP32 — FP16 destroys RF-DETR's transformer decoder per prior calibration
network-mode=0

# 2 = person + handbag (training indices 0 and 1).
# DeepStreamOutput fuses RFDETR logits → argmax class_id. Queries that
# don't match either class fall into the extra "no-object" channel and
# are filtered by the NMS post-processor via pre-cluster-threshold.
num-detected-classes=2

interval=0
gie-unique-id=1

# DeepStream-Yolo fused output head appended by onnx_to_engine.py
output-blob-names=output
network-type=0
parse-bbox-func-name=NvDsInferParseYolo
custom-lib-path=/app/nvdsinfer_custom_impl_Yolo/libnvdsinfer_custom_impl_Yolo.so

# cluster-mode=4 (NONE), not 2 (NMS): the fused output head above already
# emits final, per-query decoded (x1,y1,x2,y2,score,class_id) boxes —
# RF-DETR is query-based (DETR), not a dense-anchor detector, so there is
# no raw anchor grid left for NMS to cluster. Running nvinfer's NMS-based
# clustering on already-final boxes is exactly the documented DeepStream
# failure mode for a post-NMS [N,300,6]-style output: boxes render
# visibly rotated/shifted (~45°/135°) because the clustering step
# misinterprets already-decoded coordinates as pre-NMS anchor data.
cluster-mode=4
maintain-aspect-ratio=0

[class-attrs-all]
pre-cluster-threshold=0.40
topk=100

# Person (0) — allow weaker tracks so tracker's hysteresis picks them up
[class-attrs-0]
pre-cluster-threshold=0.30
topk=50

# Handbag (1) — raised to 0.40 to kill shadow false positives.
# Fine-tune auto-labels (threshold 0.05) leaked shadow-blob "handbags"
# into training; raising deploy threshold masks that at the cost of
# some dumped-bag recall. Next iteration: retrain with cleaner labels
# (auto-label at 0.20 + add hard negatives from shadow frames).
[class-attrs-1]
pre-cluster-threshold=0.40
