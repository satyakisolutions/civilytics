# WasteWatch Primary GIE — RF-DETR Object Detection
#
# Classes follow the COCO 91-index convention (NOT COCO80). RF-DETR's
# exported ONNX has a 91-dim logits head, so after the DeepStreamOutput
# fusion (scripts/onnx_to_engine.py) the class_id emitted in slot 5 of
# every row of the [300, 6] output tensor is in [0..90], where 0 is
# "background". Keep num-detected-classes + labelfile-path in lock-step
# with this — otherwise NvDsInferParseYolo reads preclusterThreshold
# out of bounds when class_id >= num-detected-classes.
#
# ---------------- Threshold calibration ----------------
# RF-DETR is sigmoid-head DETR; healthy scores on a real daylight
# frame from the LSG footage (validated offline via
# scripts/test_rfdetr_inference.py on VID-20260131-WA0007.mp4):
#
#   - motorcycle (cls 4)   0.85
#   - car        (cls 3)   0.79
#   - person     (cls 1)   0.48
#   - truck      (cls 8)   0.38
#
# Earlier "all scores stuck at 0.06" measurements were caused by an
# opset-13 ReduceSum being rewritten to the attribute form, which
# broke the decoder cross-attention and turned every query into
# noise. That surgery is now guarded by `ModelSpec.convert_reduce_axes`
# (off for RF-DETR, on for the action classifier).
#
# 0.25 gives a clean precision floor for person/vehicle classes, and
# 0.20 for carried objects lets the tracker + FSM promote weaker
# evidence only when dwell time confirms it.
#
# Copyright (c) 2024-2026 Satyaki Solutions Pvt Ltd. All rights reserved.

[property]
gpu-id=0
net-scale-factor=0.00392156862745

# RF-DETR base was exported at 560x560 with static batch=1 (see
# pipeline/app/engine_builder.py for the matching TRT min/opt/max
# shapes). Keep nvinfer and the engine spec in lock-step or DeepStream
# will silently resize/crop and confuse the position-embedding layer.
infer-dims=3;560;560

# Absolute paths — DeepStream resolves relative model-engine-file
# relative to the nvinfer config file's directory (/app/configs),
# not the pipeline cwd. Using absolute /app/models/... keeps both
# the bind-mount override and the baked-in fallback working.
model-engine-file=/app/models/rfdetr/rfdetr_fp16.engine
onnx-file=/app/models/rfdetr/rfdetr.onnx
labelfile-path=/app/configs/labels.txt

batch-size=1
process-mode=1
model-color-format=0
network-mode=2

# 91 = COCO full index range (including background at index 0).
# MUST equal the number of lines in labels.txt.
num-detected-classes=91

# Run inference on every frame. interval>0 leaves the in-between
# frames with no obj_meta (the tracker's Kalman predictions are not
# inserted into obj_meta_list by default), which presents on the
# output stream as bboxes flickering on/off every 1-3 frames. RF-DETR
# FP16 is ~60 ms/frame on a 3060 — fine for 2 sources at ~15 fps each
# in the current LSG demo.
interval=0
gie-unique-id=1

# ---- Post-processing ----
# RF-DETR exports two tensors natively (`labels` [N,Q,C] + `dets` [N,Q,4]);
# scripts/onnx_to_engine.py appends a DeepStreamOutput head via ONNX
# graphsurgeon that fuses them into a single YOLO-compatible layout
#   output  [N, Q, 6]  where each row is (x1, y1, x2, y2, score, class_id)
# Those bboxes are then decoded by marcoslucianops' NvDsInferParseYolo
# parser compiled into libnvdsinfer_custom_impl_Yolo.so (see
# pipeline/nvdsinfer_custom_impl_Yolo/Makefile).
output-blob-names=output
network-type=0
parse-bbox-func-name=NvDsInferParseYolo
custom-lib-path=/app/nvdsinfer_custom_impl_Yolo/libnvdsinfer_custom_impl_Yolo.so

# cluster-mode=2 == Non-Maximum Suppression. The DETR loss only dedupes
# queries during training; at inference RF-DETR's 300 queries still
# frequently converge multiple proposals on the same object (5-30
# near-duplicate boxes per real target in our testing). Without NMS
# every duplicate passes through and the OSD renders a "100 bboxes
# per scooter" wall of noise. Per-class nms-iou-threshold below is
# what actually collapses the duplicates.
cluster-mode=2

# RF-DETR does not resize with padding — disable maintain-aspect-ratio
# per the upstream guidance in DeepStream-Yolo/docs/RFDETR.md.
maintain-aspect-ratio=0

# ---- Per-class pre-cluster thresholds ----
# Default for "we don't care about this class here" — still low enough
# to actually be exceeded, just higher than the interesting classes so
# the noise gets filtered out first. Background remains suppressed.
[class-attrs-all]
pre-cluster-threshold=0.08
topk=100
nms-iou-threshold=0.5
